// ==========================================
// E-Commerce Database
// Stage 1: Foundation & Users
// ==========================================


// =====================
// Users
// =====================

Table users {
  id integer [pk, increment]
  first_name varchar
  last_name varchar
  mobile varchar [unique]
  password varchar
  status varchar
  mobile_verified_at timestamp
  created_at timestamp
  updated_at timestamp
}


// =====================
// Roles
// =====================

Table roles {
  id integer [pk, increment]
  name varchar [unique]
  created_at timestamp
}


// =====================
// Permissions
// =====================

Table permissions {
  id integer [pk, increment]
  name varchar [unique]
  created_at timestamp
}


// =====================
// Role User
// Many-to-Many
// =====================

Table role_user {
  role_id integer [not null]
  user_id integer [not null]

  indexes {
    (role_id, user_id) [pk]
  }
}


// =====================
// Permission Role
// Many-to-Many
// =====================

Table role_permission {
  
  role_id integer [not null]
  permission_id integer [not null]

  indexes {
    (permission_id, role_id) [pk]
  }
}


// ==========================================
// Relationships
// ==========================================

// User <-> Role
Ref: role_user.role_id > roles.id
Ref: role_user.user_id > users.id

// Role <-> Permission
Ref: role_permission.permission_id > permissions.id
Ref: role_permission.role_id > roles.id

// =====================
// Addresses
// =====================

Table addresses {
  id integer [pk, increment]
  user_id integer [not null]
  title varchar
  recipient_name varchar
  mobile varchar
  province varchar
  city varchar
  postal_code varchar
  address text
  is_default boolean
  created_at timestamp
  updated_at timestamp
}


// ==========================================
// Address Relationships
// ==========================================

Ref: users.id < addresses.user_id


// =====================
// Categories
// =====================

Table categories {
  id integer [pk, increment]
  parent_id integer
  name varchar
  slug varchar [unique]
  description text
  image varchar
  status varchar
  created_at timestamp
  updated_at timestamp
}


// =====================
// Brands
// =====================

Table brands {
  id integer [pk, increment]
  name varchar
  slug varchar [unique]
  description text
  logo varchar
  status varchar
  created_at timestamp
  updated_at timestamp
}


// =====================
// Products
// =====================

Table products {
  id integer [pk, increment]
  brand_id integer
  name varchar
  slug varchar [unique]
  sku varchar [unique]
  description text
  short_description text
  type varchar
  status varchar
  created_at timestamp
  updated_at timestamp
}


// =====================
// Product Categories
// Many-to-Many
// =====================

Table product_categories {
  product_id integer [not null]
  category_id integer [not null]

  indexes {
    (product_id, category_id) [pk]
  }
}


// =====================
// Product Variants
// =====================

Table product_variants {
  id integer [pk, increment]
  product_id integer [not null]
  sku varchar [unique]
  price decimal
  compare_price decimal
  cost_price decimal
  status varchar
  created_at timestamp
  updated_at timestamp
}


// =====================
// Product Images
// =====================

Table product_images {
  id integer [pk, increment]
  product_id integer [not null]
  variant_id integer
  image varchar
  sort_order integer
  is_primary boolean
  created_at timestamp
}


// =====================
// Attributes
// =====================

Table attributes {
  id integer [pk, increment]
  name varchar
  slug varchar [unique]
  type varchar
  created_at timestamp
  updated_at timestamp
}


// =====================
// Attribute Values
// =====================

Table attribute_values {
  id integer [pk, increment]
  attribute_id integer [not null]
  value varchar
  slug varchar
  created_at timestamp
  updated_at timestamp
}


// =====================
// Product Attributes
// =====================

Table product_attributes {
  product_id integer [not null]
  attribute_id integer [not null]
  attribute_value_id integer [not null]

  indexes {
    (product_id, attribute_id, attribute_value_id) [pk]
  }
}


// =====================
// Product Variant Values
// =====================

Table product_variant_values {
  product_variant_id integer [not null]
  attribute_value_id integer [not null]

  indexes {
    (product_variant_id, attribute_value_id) [pk]
  }
}


// ==========================================
// Relationships
// ==========================================

// Categories
Ref: categories.parent_id > categories.id

// Brand -> Products
Ref: brands.id < products.brand_id

// Products <-> Categories
Ref: products.id < product_categories.product_id
Ref: categories.id < product_categories.category_id

// Product -> Variants
Ref: products.id < product_variants.product_id

// Product -> Images
Ref: products.id < product_images.product_id

// Variant -> Images
Ref: product_variants.id < product_images.variant_id

// Attributes
Ref: attributes.id < attribute_values.attribute_id

// Product Attributes
Ref: products.id < product_attributes.product_id
Ref: attributes.id < product_attributes.attribute_id
Ref: attribute_values.id < product_attributes.attribute_value_id

// Variant Values
Ref: product_variants.id < product_variant_values.product_variant_id
Ref: attribute_values.id < product_variant_values.attribute_value_id

Table warehouses {
  id integer [pk, increment]
  name varchar
  code varchar [unique]
  address text
  status varchar
  created_at timestamp
  updated_at timestamp
}

Table inventories {
  id integer [pk, increment]
  warehouse_id integer [not null]
  product_variant_id integer [not null]
  quantity integer
  reserved_quantity integer
  low_stock_threshold integer
  created_at timestamp
  updated_at timestamp

  indexes {
    (warehouse_id, product_variant_id) [unique]
  }
}

Table inventory_transactions {
  id integer [pk, increment]
  inventory_id integer [not null]
  type varchar
  quantity integer
  reference_type varchar
  reference_id integer
  note text
  created_at timestamp
}

Ref: warehouses.id < inventories.warehouse_id
Ref: product_variants.id < inventories.product_variant_id
Ref: inventories.id < inventory_transactions.inventory_id

Table payments {
  id integer [pk, increment]
  order_id integer [not null]
  amount decimal [not null]
  method varchar
  status varchar
  transaction_id varchar
  gateway varchar
  paid_at timestamp
  created_at timestamp
  updated_at timestamp
}

Table wallets {
  id integer [pk, increment]
  user_id integer [not null, unique]
  balance decimal
  status varchar
  created_at timestamp
  updated_at timestamp
}

Table wallet_transactions {
  id integer [pk, increment]
  wallet_id integer [not null]
  type varchar
  direction varchar
  amount decimal
  balance_before decimal
  balance_after decimal
  reference_type varchar
  reference_id integer
  description text
  created_at timestamp
}

Ref: orders.id < payments.order_id
Ref: users.id - wallets.user_id
Ref: wallets.id < wallet_transactions.wallet_id
